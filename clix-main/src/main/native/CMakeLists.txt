cmake_minimum_required(VERSION 3.15)
project(clix_launcher C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(clix clix_launcher.c)
if(MSVC)
  target_compile_definitions(clix PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(clix kernel32.lib)
  target_link_options(clix PRIVATE "/SUBSYSTEM:CONSOLE")
else()
  target_link_libraries(clix kernel32)
  # wmain needs the Unicode entry point when building with MinGW-w64.
  target_link_options(clix PRIVATE -municode)
endif()
